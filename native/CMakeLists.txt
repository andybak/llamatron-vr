cmake_minimum_required(VERSION 3.21)
project(LlamatronNative LANGUAGES CXX RC)
add_executable(LlamatronNative main.cpp capture.cpp renderer.cpp xr_session.cpp generated/loader.rc)
target_compile_features(LlamatronNative PRIVATE cxx_std_20)
target_compile_definitions(LlamatronNative PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11 XR_NO_PROTOTYPES)
target_include_directories(LlamatronNative PRIVATE vendor generated)
target_link_libraries(LlamatronNative PRIVATE d3d11 dxgi d3dcompiler windowscodecs ole32 shell32)
set_property(TARGET LlamatronNative PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(LlamatronNative PRIVATE /W4 /EHsc)
endif()
